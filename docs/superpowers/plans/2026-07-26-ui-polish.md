```ts
// True workout duration: sessions get opened before the warm-up, so the
// clock starts at the first logged set, not at session creation. HYROX has
// its own timeline — the summed segments are the workout.
export function sessionDurationMin(s: {
  type: string;
  startedAt: Date | null;
  endedAt: Date | null;
  firstSetAt: Date | null;
  segmentsMs: number;
}): number | null {
  if (s.type === "HYROX") {
    if (s.segmentsMs > 0) return Math.max(1, Math.round(s.segmentsMs / 60_000));
    // fall through to wall-clock for legacy hyrox sessions without segments
  }
  if (s.endedAt === null) return null;
  const start = s.firstSetAt ?? s.startedAt;
  if (start === null) return null;
  const min = Math.round((s.endedAt.getTime() - start.getTime()) / 60_000);
  return min >= 0 ? Math.max(1, min) : null;
}
```

```ts
// min(sets.createdAt) and sum(segments.durationMs) per session, one query each.
export async function attachDurationInputs(athleteId: string, sessionIds: string[]) {
  if (sessionIds.length === 0) return new Map<string, { firstSetAt: Date | null; segmentsMs: number }>();
  const firstSets = await db
    .select({
      sessionId: sessionBlocks.sessionId,
      firstSetAt: sql<Date>`MIN(${sets.createdAt})`.mapWith((v) => new Date(v)),
    })
    .from(sets)
    .innerJoin(blockMovements, eq(sets.blockMovementId, blockMovements.id))
    .innerJoin(sessionBlocks, eq(blockMovements.blockId, sessionBlocks.id))
    .where(and(eq(sets.athleteId, athleteId), inArray(sessionBlocks.sessionId, sessionIds)))
    .groupBy(sessionBlocks.sessionId);
  const segs = await db
    .select({
      sessionId: sessionSegments.sessionId,
      segmentsMs: sql<number>`COALESCE(SUM(${sessionSegments.durationMs}), 0)::int`,
    })
    .from(sessionSegments)
    .where(and(eq(sessionSegments.athleteId, athleteId), inArray(sessionSegments.sessionId, sessionIds)))
    .groupBy(sessionSegments.sessionId);
  const map = new Map<string, { firstSetAt: Date | null; segmentsMs: number }>();
  for (const id of sessionIds) map.set(id, { firstSetAt: null, segmentsMs: 0 });
  for (const r of firstSets) map.set(r.sessionId, { ...(map.get(r.sessionId) as { firstSetAt: Date | null; segmentsMs: number }), firstSetAt: r.firstSetAt });
  for (const r of segs) map.set(r.sessionId, { ...(map.get(r.sessionId) as { firstSetAt: Date | null; segmentsMs: number }), segmentsMs: r.segmentsMs });
  return map;
}
```

```ts
// Boxing-ring gong: low fundamental + inharmonic partials with slow decay.
function gongStrike(audio: AudioContext, at: number, gain = 0.5) {
  const partials: [number, number][] = [
    [180, 1],
    [277, 0.6],
    [412, 0.35],
    [739, 0.2],
  ];
  for (const [freq, amp] of partials) {
    const osc = audio.createOscillator();
    const g = audio.createGain();
    osc.type = "sine";
    osc.frequency.setValueAtTime(freq, audio.currentTime + at);
    g.gain.setValueAtTime(gain * amp, audio.currentTime + at);
    g.gain.exponentialRampToValueAtTime(0.001, audio.currentTime + at + 1.4);
    osc.connect(g).connect(audio.destination);
    osc.start(audio.currentTime + at);
    osc.stop(audio.currentTime + at + 1.5);
  }
}
```
